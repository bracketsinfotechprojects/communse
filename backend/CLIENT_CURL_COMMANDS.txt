===============================================
BASIC COMMANDS
===============================================

1. GENERATE CHAT TOKEN (GET):
curl -X GET "http://localhost:3000/chat/token?userId=USER_ID&eventId=EVENT_ID" \
  -H "Content-Type: application/json" \
  -H "Accept: application/json"

2. GENERATE CHAT TOKEN (POST):
curl -X POST "http://localhost:3000/chat/token" \
  -H "Content-Type: application/json" \
  -H "Accept: application/json" \
  -d '{"userId":"USER_ID","eventId":"EVENT_ID"}'

3. VERIFY TOKEN:
curl -X GET "http://localhost:3000/chat/verify?token=TOKEN_HERE" \
  -H "Content-Type: application/json" \
  -H "Accept: application/json"

4. CHECK PERMISSION:
curl -X POST "http://localhost:3000/chat/permission" \
  -H "Content-Type: application/json" \
  -H "Accept: application/json" \
  -d '{"userId":"USER_ID","eventId":"EVENT_ID"}'

===============================================
CLIENT-SIDE JAVASCRIPT EXAMPLE
===============================================

async function getChatToken(userId, eventId) {
  try {
    const response = await fetch(
      `http://localhost:3000/chat/token?userId=${userId}&eventId=${eventId}`,
      {
        method: 'GET',
        headers: {
          'Content-Type': 'application/json',
          'Accept': 'application/json'
        }
      }
    );
    
    const data = await response.json();
    
    if (data.success) {
      // Use token with Firebase Client SDK
      const { signInWithCustomToken } = await import('firebase/auth');
      const { auth } = await import('./firebase-config');
      
      const userCredential = await signInWithCustomToken(auth, data.token);
      console.log('User signed in for chat:', userCredential.user.uid);
      
      return data.token;
    } else {
      throw new Error(data.error);
    }
  } catch (error) {
    console.error('Chat token error:', error);
    throw error;
  }
}
