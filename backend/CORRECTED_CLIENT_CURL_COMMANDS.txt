CLIENT-SIDE JAVASCRIPT EXAMPLE
===============================================

async function getChatToken(userId, eventId) {
  try {
    const response = await fetch(
      `http://localhost:5000/chat/token?userId=${userId}&eventId=${eventId}`,
      {
        method: 'GET',
        headers: {
          'Content-Type': 'application/json',
          'Accept': 'application/json'
        }
      }
    );
    
    const data = await response.json();
    
    if (data.success) {
      console.log('Token received:', data.token);
      console.log('User role:', data.claims.role);
      
      // Use token with Firebase Client SDK
      const { signInWithCustomToken } = await import('firebase/auth');
      const { auth } = await import('./firebase-config');
      
      const userCredential = await signInWithCustomToken(auth, data.token);
      console.log('User signed in for chat:', userCredential.user.uid);
      
      return data.token;
    } else {
      throw new Error(data.error);
    }
  } catch (error) {
    console.error('Chat token error:', error);
    throw error;
  }
}
